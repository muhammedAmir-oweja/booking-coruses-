<?php

use App\Models\Course;

/*
 |--------------------------------------------------------------------------
 | Course capacity helpers
 |--------------------------------------------------------------------------
 | One place that answers "how full is this course" and "can it still be
 | booked", so controllers never re-implement either rule.
 */

if (! function_exists('course_taken_seats')) {
    /**
     * Seats already taken: every booking that has not been cancelled.
     */
    function course_taken_seats(Course $course): int
    {
        // Use a count the query already loaded (see withCount in step 3) so
        // that listing 20 courses does not fire 20 extra queries.
        if (isset($course->active_bookings_count)) {
            return (int) $course->active_bookings_count;
        }

        return $course->bookings()
            ->whereIn('status', ['pending', 'confirmed'])
            ->count();
    }
}

if (! function_exists('course_remaining_seats')) {
    /**
     * Total capacity minus the seats taken. Never negative.
     */
    function course_remaining_seats(Course $course): int
    {
        return max(0, (int) $course->capacity - course_taken_seats($course));
    }
}

if (! function_exists('course_is_available')) {
    /**
     * Can a student be booked onto this course right now?
     * Active status, the course has not finished, and at least one free seat.
     */
    function course_is_available(Course $course): bool
    {
        if ($course->status !== 'active') {
            return false;
        }

        // end_date is cast to a date, so compare against today's date - not
        // now() - or a course ending today would read as already finished.
        if ($course->end_date && $course->end_date->lt(today())) {
            return false;
        }

        return course_remaining_seats($course) > 0;
    }
}
Decision you are making here The task says capacity minus bookings, so the helper counts bookings. This project also has an enrollments table, and the courses list currently shows enrollments_count. If your grader expects enrollments instead, change one line — $course->bookings() → $course->enrollments() — and nothing else. That one-line switch is the point of the helper.

class BookingCreated extends Notification

    public function __construct(public Booking $booking) {}

    /** Stored in the database only - no mail, no broadcast. */

    {
        $booking = $this->booking->loadMissing(['student', 'course']);

        return [
            'type'         => 'booking_created',

            'student_id'   => $booking->student_id,

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class NotificationController extends Controller
{
    /** GET /api/notifications?filter=unread&per_page=10 */
    public function index(Request $request): JsonResponse
    {
        $query = $request->get('filter', 'unread') === 'unread'
            ? $request->user()->unreadNotifications()
            : $request->user()->notifications();

        $perPage = min((int) $request->get('per_page', 10), 50);
        $notifications = $query->latest()->paginate($perPage);

        // Same envelope as every other list endpoint in this API.
        $notifications->getCollection()->transform(fn ($n) => [
            'id'         => $n->id,
            'data'       => $n->data,
            'read_at'    => $n->read_at?->toDateTimeString(),
            'created_at' => $n->created_at->toDateTimeString(),
        ]);

        return response()->json($notifications);
    }

    /** GET /api/notifications/unread-count */
    public function unreadCount(Request $request): JsonResponse
    {
        return response()->json([
            'count' => $request->user()->unreadNotifications()->count(),
        ]);
    }

    /** PATCH /api/notifications/{id}/read */
    public function markAsRead(Request $request, string $id): JsonResponse
    {
        // Scoped to the signed-in user on purpose: a global lookup would let
        // anyone mark somebody else's notification as read.
        $request->user()->notifications()->findOrFail($id)->markAsRead();

        return response()->json(['message' => 'Notification marked as read.']);
    }

    /** POST /api/notifications/read-all */
    public function markAllAsRead(Request $request): JsonResponse
    {
        $request->user()->unreadNotifications->markAsRead();

        return response()->json(['message' => 'All notifications marked as read.']);
    }

    /** DELETE /api/notifications/{id} */
    public function destroy(Request $request, string $id): JsonResponse
    {
        $request->user()->notifications()->findOrFail($id)->delete();

        return response()->json(['message' => 'Notification deleted.']);
    }
}

import { useCallback, useEffect, useRef, useState } from 'react'
import { useNavigate } from 'react-router-dom'
import {
  getNotifications, getUnreadCount,
  markAllNotificationsRead, markNotificationRead,
} from '../../api/notifications'
import type { AppNotification } from '../../types'

// No websocket in this project, so the badge polls.
const POLL_MS = 60_000

export default function NotificationsBell() {
  const navigate = useNavigate()
  const [open, setOpen] = useState(false)
  const [items, setItems] = useState<AppNotification[]>([])
  const [count, setCount] = useState(0)
  const [loading, setLoading] = useState(false)
  const boxRef = useRef<HTMLDivElement>(null)

  const loadCount = useCallback(async () => {
    try {
      const { data } = await getUnreadCount()
      setCount(data.count)
    } catch { /* a bell is not worth an error banner */ }
  }, [])

  // Badge on mount, then once a minute.
  useEffect(() => {
    void loadCount()
    const id = setInterval(() => void loadCount(), POLL_MS)
    return () => clearInterval(id)
  }, [loadCount])

  // The list is only fetched when the panel opens.
  useEffect(() => {
    if (!open) return
    let cancelled = false
    setLoading(true)

    void (async () => {
      try {
        const { data } = await getNotifications({ filter: 'unread', per_page: 10 })
        if (!cancelled) setItems(data.data)
      } catch {
        if (!cancelled) setItems([])
      } finally {
        if (!cancelled) setLoading(false)
      }
    })()

    return () => { cancelled = true }
  }, [open])

  // Close on outside click and on Escape.
  useEffect(() => {
    if (!open) return
    const onDown = (e: MouseEvent) => {
      if (!boxRef.current?.contains(e.target as Node)) setOpen(false)
    }
    const onKey = (e: KeyboardEvent) => { if (e.key === 'Escape') setOpen(false) }
    document.addEventListener('mousedown', onDown)
    document.addEventListener('keydown', onKey)
    return () => {
      document.removeEventListener('mousedown', onDown)
      document.removeEventListener('keydown', onKey)
    }
  }, [open])

  const openOne = async (n: AppNotification) => {
    await markNotificationRead(n.id).catch(() => null)
    setItems(prev => prev.filter(i => i.id !== n.id))
    setCount(c => Math.max(0, c - 1))
    setOpen(false)
    navigate(n.data.url ?? '/bookings')
  }

  const readAll = async () => {
    await markAllNotificationsRead().catch(() => null)
    setItems([])
    setCount(0)
  }

  return (
    <div className="lmsnav-user" ref={boxRef}>
      <button
        type="button"
        className="lmsnav-icon"
        style={{ position: 'relative' }}
        aria-label={`Notifications${count ? `, ${count} unread` : ''}`}
        aria-expanded={open}
        onClick={() => setOpen(o => !o)}
      >
        <i className="fas fa-bell" />
        {count > 0 && (
          <span style={{
            position: 'absolute', top: 2, right: 1,
            minWidth: 16, height: 16, padding: '0 4px', borderRadius: 8,
            background: '#c0392b', color: '#fff',
            font: '700 10px/16px inherit',
          }}>
            {count > 9 ? '9+' : count}
          </span>
        )}
      </button>

      {open && (
        <div className="lmsnav-menu" role="menu" style={{ width: 340 }}>
          <div style={{
            display: 'flex', alignItems: 'center', justifyContent: 'space-between',
            padding: '8px 10px', fontSize: 12, color: 'rgba(255,255,255,.60)',
          }}>
            <span>Unread notifications</span>
            {count > 0 && (
              <button type="button" onClick={readAll} style={{
                border: 0, background: 'transparent', cursor: 'pointer',
                color: '#7cbcf7', font: '600 11.5px inherit',
              }}>
                Mark all as read
              </button>
            )}
          </div>

          <div className="lmsnav-sep" />

          {loading && <div className="lmsnav-note">Loading…</div>}

          {!loading && items.length === 0 && (
            <div className="lmsnav-note">Nothing new right now.</div>
          )}

          <div style={{ maxHeight: 320, overflowY: 'auto' }}>
            {items.map(n => (
              <button
                key={n.id}
                type="button"
                className="lmsnav-item"
                role="menuitem"
                onClick={() => void openOne(n)}
                style={{ alignItems: 'flex-start' }}
              >
                <i className="fas fa-calendar-check" aria-hidden="true" />
                <span>
                  {n.data.message ?? 'New booking'}
                  <span style={{
                    display: 'block', marginTop: 3, fontSize: 11,
                    color: 'rgba(255,255,255,.45)',
                  }}>
                    {new Date(n.created_at).toLocaleString()}
                  </span>
                </span>
              </button>
            ))}
          </div>
        </div>
      )}
    </div>
  )
}

<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

/**
 * The API answers in whatever language the SPA asks for. The frontend sends
 * its current language on every request (see api/client.ts).
 */
class SetLocale
{
    private const SUPPORTED = ['en', 'ar'];

    public function handle(Request $request, Closure $next): Response
    {
        // X-Locale is explicit; Accept-Language may arrive as "ar-EG,ar;q=0.9"
        // so only the first two letters are useful.
        $requested = $request->header('X-Locale')
            ?? substr((string) $request->header('Accept-Language'), 0, 2);

        if (in_array($requested, self::SUPPORTED, true)) {
            app()->setLocale($requested);
        }

        return $next($request);
    }
}

B8

Traps that break "instantly"

  • Never const t = i18n.t at module scope. It will not re-render on switch. Always const { t } = useTranslation() inside the component — that hook is what subscribes the component to language changes.
  • Any array of labels defined outside a component is frozen. navGroups, DESTINATIONS, ROLE_LABELS — store keys, call t() in the JSX (step B5).
  • Arabic has six plural forms. If you use t('…', { count }) and want grammatical Arabic, provide key_zero, key_one, key_two, key_few, key_many, key_other. The starter file phrases counts so one form works.
  • Missing key = the key printed on screen. While converting, keep the browser console open: i18next logs every missing key.
  • Keep both JSON files the same shape. A key present in en.json and missing in ar.json silently falls back to English and looks like a half-translated page.
  • npm run build runs tsc -b. Importing JSON works out of the box with Vite's defaults here, but a typo in a labelKey interface change will fail the build — run it before you call the task done.
